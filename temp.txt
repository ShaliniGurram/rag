def classify_graph(points):
        xs = [p['data_x'] for p in points]
        ys = [p['data_y'] for p in points]
        n = len(points)
        if n < 4:
            return "unknown"
        
        xs_sorted, ys_sorted = zip(*sorted(zip(xs, ys)))
        dy = np.diff(ys_sorted)
        dx = np.diff(xs_sorted)

        slope_changes = np.count_nonzero(np.diff(np.sign(dy)))
        
        if slope_changes == 0:
            return "linear increasing" if dy.mean() > 0 else "linear decreasing"
        elif slope_changes <= n * 0.2:
            return "smooth curve"
        elif n < 15:
            return "scatter"
        else:
            return "oscillating / non-linear"
def analyze_series(series):
    xs = np.array([p['data_x'] for p in series['points']])
    ys = np.array([p['data_y'] for p in series['points']])
    # remove extreme outliers using percentiles
    if len(xs) >= 4:
        keep_mask = (xs > np.percentile(xs, 1)) & (xs < np.percentile(xs, 99)) & (ys > np.percentile(ys, 1)) & (ys < np.percentile(ys, 99))
        xs = xs[keep_mask]
        ys = ys[keep_mask]
    if len(xs) == 0:
        return {'min_x': 0., 'max_x': 0., 'min_y': 0., 'max_y': 0., 'mean_y': 0., 'std_y': 0., 'trend': 'unknown'}
    order = np.argsort(xs)
    xs_sorted = xs[order]
    ys_sorted = ys[order]
    slope = np.polyfit(xs_sorted, ys_sorted, 1)[0] if len(xs_sorted) > 1 else 0.0
    stats = {
        'min_x': float(xs_sorted.min()),
        'max_x': float(xs_sorted.max()),
        'min_y': float(ys_sorted.min()),
        'max_y': float(ys_sorted.max()),
        'mean_y': float(ys_sorted.mean()),
        'std_y': float(ys_sorted.std()),
        'trend': "increasing" if slope > 0 else "decreasing"
    }
    return stats

def summarize_series(series):
    """Create a short human-readable summary."""
    s = series['stats']
    return (
        f"Series {series['series_label']} ({series['type']}): "
        f"Values range from {s['min_y']:.2f} to {s['max_y']:.2f}, "
        f"mean value {s['mean_y']:.2f}. Overall trend is {s['trend']}."
    )
class PDFExtractor:
    def __init__(self, pdf_path, output_dir="extracted_content", ocr_dpi=200, graph_x_range = None, graph_y_range = None):
        self.pdf_path = pdf_path
        self.output_dir = Path(output_dir)
        self.ocr_dpi = ocr_dpi
        self._setup_tesseract()
        # Default numeric ranges for graphs (used when OCR of ticks is not available)
        self.graph_x_range = graph_x_range
        self.graph_y_range = graph_y_range

        self.extracted_data = {
            'text': [],
            'tables': [],
            'images': [],      
            'graphs': [],
            'ocr_text': [],    
            'metadata': {}
        }
        self.all_content = []

        (self.output_dir / "images").mkdir(parents=True, exist_ok=True)
        (self.output_dir / "pages").mkdir(parents=True, exist_ok=True)

    def extract_all(self):
        print(f"Processing PDF: {self.pdf_path}")
        self.extract_metadata()
        self.extract_text_and_tables()
        self.extract_images_and_ocr()
        print("\nExtraction complete!")
        self.print_summary()
        return self.extracted_data

    def extract_metadata(self):
        with open(self.pdf_path, 'rb') as file:
            reader = PyPDF2.PdfReader(file)
            info = dict(reader.metadata) if reader.metadata else {}
            self.extracted_data['metadata'] = {
                'pages': len(reader.pages),
                'info': info,
            }

    def extract_text_and_tables(self):
        with pdfplumber.open(self.pdf_path) as pdf:
            for page_num, page in enumerate(pdf.pages, 1):
                text = page.extract_text(
                    x_tolerance=1.5,
                    y_tolerance=2,
                    keep_blank_chars=False,
                    use_text_flow=True
                )
                if text:
                    self.extracted_data['text'].append({
                        'page': page_num,
                        'content': text
                    })
                tables = page.extract_tables()
                if tables:
                    for table_idx, table in enumerate(tables, 1):
                        if table and len(table) > 0:
                            if len(table) > 1:
                                df = pd.DataFrame(table[1:], columns=table[0])
                            else:
                                df = pd.DataFrame(table)
                            self.extracted_data['tables'].append({
                                'page': page_num,
                                'table_num': table_idx,
                                'dataframe': df,
                                'raw': table
                            })

    def extract_images_and_ocr(self):
        try:
            print("\nExtracting embedded images and page OCR with PyMuPDF...")
            doc = fitz.open(self.pdf_path)
            img_dir = self.output_dir / "images"
            page_dir = self.output_dir / "pages"
            graphs_dir = self.output_dir / "graphs"
            graphs_dir.mkdir(parents=True, exist_ok=True)

            for page_idx in range(len(doc)):
                page = doc.load_page(page_idx)

                # ----------- Extract Embedded Images -----------
                image_list = page.get_images(full=True)
                for img_index, img_info in enumerate(image_list, start=1):
                    xref = img_info[0]
                    base_image = doc.extract_image(xref)
                    image_bytes = base_image["image"]
                    image_ext = base_image.get("ext", "png")
                    width = base_image.get("width")
                    height = base_image.get("height")

                    image_filename = f"page{page_idx + 1}_img{img_index}.{image_ext}"
                    image_path = img_dir / image_filename
                    with open(image_path, "wb") as img_file:
                        img_file.write(image_bytes)

                    self.extracted_data['images'].append({
                        'page': page_idx + 1,
                        'filename': str(image_path),
                        'size': (width, height),
                        'xref': xref
                    })

                    # Analyze graph from the embedded image
                    self._process_graph_from_image(image_path, page_idx, graphs_dir)

                # ----------- Extract Full Page OCR -----------
                pix = page.get_pixmap(dpi=self.ocr_dpi)
                if pix.alpha:
                    pix = fitz.Pixmap(fitz.csRGB, pix)
                page_img_path = page_dir / f"page_{page_idx + 1}.png"
                pix.save(str(page_img_path))

                try:
                    with Image.open(page_img_path) as pil_img:
                        ocr_text = pytesseract.image_to_string(pil_img)
                        if ocr_text.strip():
                            self.extracted_data['ocr_text'].append({
                                'page': page_idx + 1,
                                'image': str(page_img_path),
                                'text': ocr_text
                            })

                        # Detect graphs from page image
                        self._process_graph_from_image(page_img_path, page_idx, graphs_dir)

                except pytesseract.TesseractNotFoundError:
                    print(f"\nSkipping OCR for page {page_idx + 1} - Tesseract not available")

            doc.close()
            print("Image extraction and OCR complete!")

        except Exception as e:
            print(f"Error extracting images/OCR: {e}")

    def _detect_graph_range(self, pil_img):
        """
        Detect approximate X and Y axis numeric ranges using OCR from cropped axis areas.
        Returns (x_range, y_range) as tuple of (min, max).
        """
        import re
        w, h = pil_img.size

        # Crop small tight regions: bottom for X ticks, left for Y ticks.
        bottom_crop = pil_img.crop((int(w*0.08), int(h*0.82), int(w*0.95), int(h*0.97)))
        left_crop = pil_img.crop((int(w*0.02), int(h*0.08), int(w*0.18), int(h*0.95)))

        def ocr_numbers(img):
            txt = pytesseract.image_to_string(img, config='--psm 6')
            nums = re.findall(r'[-+]?\d*\.\d+|\d+', txt)
            nums = [float(n) for n in nums] if nums else []
            # remove obviously large/small outliers from OCR (e.g. page numbers)
            nums = [n for n in nums if abs(n) < 1e6] 
            return nums

        x_nums = ocr_numbers(bottom_crop)
        y_nums = ocr_numbers(left_crop)

        # Heuristic: if OCR finds 2+ numbers use min/max; else fallback to None
        x_range = (min(x_nums), max(x_nums)) if len(x_nums) >= 2 else None
        y_range = (min(y_nums), max(y_nums)) if len(y_nums) >= 2 else None

        return x_range, y_range

    def _process_graph_from_image(self, image_path, page_idx, graphs_dir):
        """Detects graphs, saves detailed analysis JSON for each page."""
        try:
            with Image.open(image_path) as pil_img:
                x_range, y_range = self._detect_graph_range(pil_img)
                series_list = self.extract_graph_points(pil_img, x_range=x_range, y_range=y_range)

                if not series_list:
                    return

                # Prepare JSON file path
                graph_json_path = graphs_dir / f"graph_analysis_page{page_idx + 1}.json"

                # Combine with existing page JSON (if any)
                if graph_json_path.exists():
                    with open(graph_json_path, "r", encoding="utf-8") as jf:
                        existing_data = json.load(jf)
                else:
                    existing_data = []

                # Append current series to existing data
                for s in series_list:
                    full_series_info = {
                        "series_id": s['series_id'],
                        "series_label": s.get("series_label", f"Series {s['series_id']}"),
                        "color": s.get("color", (0, 0, 0)),
                        "type": s.get("type", ""),
                        "stats": s.get("stats", {}),
                        "summary": s.get("summary", ""),
                        "points": s.get("points", [])  # includes px, py, data_x, data_y
                    }
                    existing_data.append(full_series_info)

                # Write JSON file (with indentation for readability)
                with open(graph_json_path, "w", encoding="utf-8") as jf:
                    json.dump(existing_data, jf, indent=4)

                # Add to extracted_data dictionary
                for s in series_list:
                    self.extracted_data['graphs'].append({
                        'page': page_idx + 1,
                        'image': str(image_path),
                        'series_id': s['series_id'],
                        'series_label': s.get('series_label', ''),
                        'color': s['color'],
                        'type': s.get('type', ''),
                        'stats': s.get('stats', {}),
                        'summary': s.get('summary', '')
                    })

        except Exception as e:
            print(f"Graph extraction failed for {image_path}: {e}")


    def extract_graph_points(self, pil_image, debug=False, x_range=None, y_range=None):
        try:
            img = cv2.cvtColor(np.array(pil_image.convert('RGB')), cv2.COLOR_RGB2BGR)
            gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
            h, w = gray.shape

            blur = cv2.GaussianBlur(gray, (5, 5), 0)
            th = cv2.adaptiveThreshold(blur, 255, cv2.ADAPTIVE_THRESH_GAUSSIAN_C,
                                       cv2.THRESH_BINARY_INV, 11, 2)
            edges = cv2.Canny(blur, 50, 150, apertureSize=3)
            lines = cv2.HoughLinesP(edges, 1, np.pi/180, threshold=80, minLineLength=int(min(img.shape[:2])*0.3), maxLineGap=20)
            mask = np.zeros_like(th)
            verticals = []
            horizontals = []
            if lines is not None:
                for x1, y1, x2, y2 in lines.reshape(-1, 4):
                    dx = abs(x2 - x1)
                    dy = abs(y2 - y1)
                    if dx < 10:
                        verticals.append((x1, y1, x2, y2))
                    if dy < 10:
                        horizontals.append((x1, y1, x2, y2))
                    cv2.line(mask, (x1, y1), (x2, y2), 255, thickness=7)
            if verticals:
                left = min(min(v[0], v[2]) for v in verticals)
                right = max(max(v[0], v[2]) for v in verticals)
            else:
                left = int(w * 0.08)
                right = int(w * 0.92)
            if horizontals:
                top = min(min(hz[1], hz[3]) for hz in horizontals)
                bottom = max(max(hz[1], hz[3]) for hz in horizontals)
            else:
                top = int(h * 0.08)
                bottom = int(h * 0.92)
            th_nolines = cv2.bitwise_and(th, cv2.bitwise_not(mask))
            kernel = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (3, 3))
            opened = cv2.morphologyEx(th_nolines, cv2.MORPH_OPEN, kernel, iterations=1)
            contours, _ = cv2.findContours(opened, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
            raw_points = []
            for cnt in contours:
                area = cv2.contourArea(cnt)
                if area < 5 or area > (w * h) * 0.01:
                    continue
                x, y, cw, ch = cv2.boundingRect(cnt)
                if cw == 0 or ch == 0:
                    continue
                aspect = float(cw) / float(ch)
                if aspect > 4 or aspect < 0.25:
                    continue
                M = cv2.moments(cnt)
                if M['m00'] == 0:
                    continue
                cX = int(M['m10'] / M['m00'])
                cY = int(M['m01'] / M['m00'])
                if cX > int(0.65 * w) and cY < int(0.18 * h):
                    continue
                if cX < int(0.06 * w) or cY > int(0.96 * h) or cY < int(0.04 * h):
                    continue
                bgr = tuple(int(v) for v in img[cY, cX])
                raw_points.append({'px': cX, 'py': cY, 'area': area, 'color': bgr})

            if not raw_points:
                return []
            clusters = []  
            def color_dist(c1, c2):
                return math.sqrt(sum((a - b) ** 2 for a, b in zip(c1, c2)))
            COLOR_THRESH = 100  
            for p in raw_points:
                assigned = False
                for cl in clusters:
                    if color_dist(p['color'], cl['center']) < COLOR_THRESH:
                        cl['points'].append(p)
                        n = len(cl['points'])
                        cl['center'] = tuple(int((cl['center'][i] * (n-1) + p['color'][i]) / n) for i in range(3))
                        assigned = True
                        break
                if not assigned:
                    clusters.append({'center': p['color'], 'points': [p]})
            MAX_CLUSTERS = 10
            while len(clusters) > MAX_CLUSTERS:
                min_d = float('inf')
                pair = (0, 1)
                for i in range(len(clusters)):
                    for j in range(i+1, len(clusters)):
                        d = color_dist(clusters[i]['center'], clusters[j]['center'])
                        if d < min_d:
                            min_d = d
                            pair = (i, j)
                i, j = pair
                ci = clusters[i]
                cj = clusters[j]
                total_pts = len(ci['points']) + len(cj['points'])
                new_center = tuple(int((ci['center'][k] * len(ci['points']) + cj['center'][k] * len(cj['points'])) / total_pts) for k in range(3))
                ci['points'].extend(cj['points'])
                ci['center'] = new_center
                clusters.pop(j)

            x0, x1 = x_range if x_range is not None else self.graph_x_range
            y0, y1 = y_range if y_range is not None else self.graph_y_range
            px_min = left
            px_max = right
            py_min = top
            py_max = bottom
            width_px = px_max - px_min if (px_max - px_min) != 0 else 1
            height_px = py_max - py_min if (py_max - py_min) != 0 else 1
            MIN_POINTS_PER_SERIES = 4
            series_list = []
            series_idx = 1
            for cl in clusters:
                if len(cl['points']) < MIN_POINTS_PER_SERIES:
                    continue
                series_points = []
                for p in cl['points']:
                    if x0 is None or x1 is None:
                        x0, x1 = (0, 1)
                    if y0 is None or y1 is None:
                        y0, y1 = (0, 1)
                    data_x = x0 + (p['px'] - px_min) / max(width_px, 1) * (x1 - x0)
                    data_y = y1 - (p['py'] - py_min) / max(height_px, 1) * (y1 - y0)
                    series_points.append({'px': p['px'], 'py': p['py'], 'area': p['area'], 'data_x': data_x, 'data_y': data_y})
                series_label = chr(ord('A') + (series_idx - 1) % 26)
                series_list.append({'series_id': series_idx, 'series_label': series_label, 'color': cl['center'], 'points': series_points})
                graph_type = classify_graph(series_points)
                stats = analyze_series({'points': series_points})
                summary = summarize_series({
                    'series_label': series_label,
                    'type': graph_type,
                    'stats': stats
                })
                series_list[-1].update({
                    'type': graph_type,
                    'stats': stats,
                    'summary': summary
                })
                series_idx += 1
            if debug:
                debug_img = img.copy()
                colors_draw = [(0,0,255),(0,255,0),(255,0,0),(0,255,255),(255,0,255),(255,255,0)]
                for si, s in enumerate(series_list):
                    col = colors_draw[si % len(colors_draw)]
                    for pt in s['points']:
                        cv2.circle(debug_img, (pt['px'], pt['py']), 4, col, -1)
                debug_out = self.output_dir / 'graphs' / 'debug_points_series.png'
                debug_out.parent.mkdir(parents=True, exist_ok=True)
                cv2.imwrite(str(debug_out), debug_img)
            return series_list
        except Exception as e:
            print(f"Error extracting graph points: {e}")
            return []

    def print_summary(self):
        print("\n" + "=" * 80)
        print("EXTRACTION SUMMARY")
        print("=" * 80)
        print(f"Total Pages: {self.extracted_data['metadata'].get('pages')}")
        print(f"Text Sections: {len(self.extracted_data['text'])}")
        print(f"Tables Found: {len(self.extracted_data['tables'])}")
        print(f"Images Extracted: {len(self.extracted_data['images'])}")
        print(f"OCR Text Blocks: {len(self.extracted_data['ocr_text'])}")
        print("=" * 80)

    def save_results(self):
        self.output_dir.mkdir(parents=True, exist_ok=True)
        with open(self.output_dir / "text.txt", 'w', encoding='utf-8') as f:
            for item in self.extracted_data['text']:
                f.write(f"=== Page {item['page']} ===\n{item['content']}\n\n")
        for item in self.extracted_data['tables']:
            filename = self.output_dir / f"table_page{item['page']}_num{item['table_num']}.csv"
            item['dataframe'].to_csv(filename, index=False)
        if self.extracted_data['ocr_text']:
            with open(self.output_dir / "ocr_text.txt", 'w', encoding='utf-8') as f:
                for item in self.extracted_data['ocr_text']:
                    f.write(f"=== Page {item['page']} - {item['image']} ===\n{item['text']}\n\n")
        if self.extracted_data['images']:
            img_rows = []
            for it in self.extracted_data['images']:
                width = None if not it['size'] else it['size'][0]
                height = None if not it['size'] else it['size'][1]
                img_rows.append({
                    'page': it['page'],
                    'filename': it['filename'],
                    'width': width,
                    'height': height,
                    'xref': it['xref']
                })
            df_imgs = pd.DataFrame(img_rows)
            df_imgs.to_csv(self.output_dir / "images_index.csv", index=False)
        print(f"\nResults saved to '{self.output_dir}/' directory")
