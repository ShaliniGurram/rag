import os
import re
from pathlib import Path
import PyPDF2
import pdfplumber
from PIL import Image
import pytesseract
import pandas as pd

class PDFExtractor:
    def __init__(self, pdf_path):
        self.pdf_path = pdf_path
        self.extracted_data = {
            'text': [],
            'tables': [],
            'images': [],
            'ocr_text': [],
            'metadata': {}
        }
        self.all_content = []
        
    def extract_all(self):
        print(f"Processing PDF: {self.pdf_path}")
        self.extract_metadata()
        self.extract_text_and_tables()
        self.extract_images()
        self.prepare_content()
        
        print("\nExtraction complete!")
        self.print_summary()
        
        return self.extracted_data
    
    def extract_metadata(self):
        with open(self.pdf_path, 'rb') as file:
            reader = PyPDF2.PdfReader(file)
            self.extracted_data['metadata'] = {
                'pages': len(reader.pages),
                'info': reader.metadata if reader.metadata else {}
            }
    
    def extract_text_and_tables(self):
        with pdfplumber.open(self.pdf_path) as pdf:
            for page_num, page in enumerate(pdf.pages, 1):
                text = page.extract_text()
                if text:
                    self.extracted_data['text'].append({
                        'page': page_num,
                        'content': text
                    })
                tables = page.extract_tables()
                if tables:
                    for table_idx, table in enumerate(tables, 1):
                        if table and len(table) > 0:
                            if len(table) > 1:
                                df = pd.DataFrame(table[1:], columns=table[0])
                            else:
                                df = pd.DataFrame(table)
                            self.extracted_data['tables'].append({
                                'page': page_num,
                                'table_num': table_idx,
                                'dataframe': df,
                                'raw': table
                            })
    
    def extract_images(self):
        with open(self.pdf_path, 'rb') as file:
            reader = PyPDF2.PdfReader(file)
            
            for page_num, page in enumerate(reader.pages, 1):
                try:
                    if '/XObject' in page['/Resources']:
                        xobjects = page['/Resources']['/XObject'].get_object()
                        
                        for obj_name in xobjects:
                            obj = xobjects[obj_name]
                            
                            if obj['/Subtype'] == '/Image':
                                try:
                                    size = (obj['/Width'], obj['/Height'])
                                    data = obj.get_data()
                                    if obj['/ColorSpace'] == '/DeviceRGB':
                                        mode = "RGB"
                                    else:
                                        mode = "P"
                                    
                                    image = Image.frombytes(mode, size, data)
                                    img_filename = f"page_{page_num}_img_{obj_name[1:]}.png"
                                    image.save(img_filename)
                                    ocr_text = pytesseract.image_to_string(image)
                                    
                                    self.extracted_data['images'].append({
                                        'page': page_num,
                                        'filename': img_filename,
                                        'size': size
                                    })
                                    if ocr_text.strip():
                                        self.extracted_data['ocr_text'].append({
                                            'page': page_num,
                                            'image': img_filename,
                                            'text': ocr_text
                                        })
                                except Exception as e:
                                    print(f"Could not extract image on page {page_num}: {e}")
                except:
                    pass
    
    def prepare_content(self):
        for item in self.extracted_data['text']:
            self.all_content.append({
                'type': 'text',
                'page': item['page'],
                'content': item['content']
            })
        for item in self.extracted_data['ocr_text']:
            self.all_content.append({
                'type': 'ocr',
                'page': item['page'],
                'image': item['image'],
                'content': item['text']
            })
        for item in self.extracted_data['tables']:
            table_text = item['dataframe'].to_string()
            self.all_content.append({
                'type': 'table',
                'page': item['page'],
                'table_num': item['table_num'],
                'content': table_text
            })
    
    def ask_question(self, question):
        keywords = question.lower().split()
        stop_words = {'what', 'is', 'are', 'the', 'a', 'an', 'in', 'on', 'at', 'to', 'for', 'of', 'and', 'or'}
        keywords = [k for k in keywords if k not in stop_words and len(k) > 2]
        
        print(f"\nQuestion: {question}")
        print(f"Searching for: {', '.join(keywords)}\n")
        print("Relevant content:")
        print("=" * 80)
        
        results = []
        for item in self.all_content:
            content_lower = item['content'].lower()
            matches = sum(1 for keyword in keywords if keyword in content_lower)
            
            if matches > 0:
                score = matches / len(keywords) if keywords else 0
                results.append((item, score))
        
        results.sort(key=lambda x: x[1], reverse=True)
        found_results = False
        for item, score in results[:5]:  
            if score > 0:
                found_results = True
                print(f"\nRelevance: {score * 100:.0f}%")
                
                if item['type'] == 'text':
                    print(f"[Page {item['page']} - Text]")
                elif item['type'] == 'ocr':
                    print(f"[Page {item['page']} - Image: {item['image']}]")
                elif item['type'] == 'table':
                    print(f"[Page {item['page']} - Table {item['table_num']}]")
                content = item['content']
                if len(content) > 500:
                    for keyword in keywords:
                        if keyword in content.lower():
                            idx = content.lower().index(keyword)
                            start = max(0, idx - 150)
                            end = min(len(content), idx + 350)
                            excerpt = content[start:end]
                            if start > 0:
                                excerpt = "..." + excerpt
                            if end < len(content):
                                excerpt = excerpt + "..."
                            print(excerpt)
                            break
                else:
                    print(content)
                
                print("-" * 80)
        
        if not found_results:
            print("No relevant content found. Try different keywords.")
        
        return results
    
    def print_summary(self):
        print("\n" + "=" * 80)
        print("EXTRACTION SUMMARY")
        print("=" * 80)
        print(f"Total Pages: {self.extracted_data['metadata']['pages']}")
        print(f"Text Sections: {len(self.extracted_data['text'])}")
        print(f"Tables Found: {len(self.extracted_data['tables'])}")
        print(f"Images Extracted: {len(self.extracted_data['images'])}")
        print(f"OCR Text Blocks: {len(self.extracted_data['ocr_text'])}")
        print("=" * 80)
    
    def save_results(self, output_dir="extracted_content"):
        os.makedirs(output_dir, exist_ok=True)
        with open(f"{output_dir}/text.txt", 'w', encoding='utf-8') as f:
            for item in self.extracted_data['text']:
                f.write(f"=== Page {item['page']} ===\n{item['content']}\n\n")
        
        for item in self.extracted_data['tables']:
            filename = f"{output_dir}/table_page{item['page']}_num{item['table_num']}.csv"
            item['dataframe'].to_csv(filename, index=False)
        if self.extracted_data['ocr_text']:
            with open(f"{output_dir}/ocr_text.txt", 'w', encoding='utf-8') as f:
                for item in self.extracted_data['ocr_text']:
                    f.write(f"=== Page {item['page']} - {item['image']} ===\n{item['text']}\n\n")
        
        print(f"\nResults saved to '{output_dir}/' directory")
